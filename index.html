function generateSidebarCategories(products) { const container = document.getElementById('sidebarCategories'); const allCats = []; products.forEach(p => { if (p.categories && Array.isArray(p.categories)) { allCats.push(...p.categories); } }); // 取得不重複的分類清單 const uniqueCats = [...new Set(allCats)]; // 🌟 如果妳希望固定某些分類在前面（例如：新加坡、伴手禮），可以這樣寫： const customOrder = ['新加坡', '伴手禮']; // 依照妳想要的順序排列 uniqueCats.sort((a, b) => { let indexA = customOrder.indexOf(a); let indexB = customOrder.indexOf(b); if (indexA === -1) indexA = 999; if (indexB === -1) indexB = 999; return indexA - indexB; }); const categories = ['全部', ...uniqueCats]; container.innerHTML = ''; categories.forEach(cat => { const btn = document.createElement('button'); btn.className = `sidebar-btn ${cat === currentCategory ? 'active' : ''}`; btn.innerText = cat; btn.onclick = () => filterCategory(cat, btn); container.appendChild(btn); }); }
