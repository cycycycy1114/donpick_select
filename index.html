import { Client } from '@notionhq/client'; export default async function handler(req, res) { res.setHeader('Content-Type', 'application/json; charset=utf-8'); if (req.method !== 'GET') { return res.status(405).json({ success: false, error: 'Method not allowed' }); } try { const notionToken = process.env.NOTION_TOKEN; const productDbId = process.env.PRODUCT_DB_ID || process.env.DATABASE_ID; const productDetailDbId = process.env.PRODUCT_DETAIL_DB_ID; const announcementDbId = process.env.ANNOUNCEMENT_DB_ID; if (!notionToken || !productDetailDbId) { return res.status(500).json({ success: false, error: '伺服器未設定 Notion 環境變數' }); } const notion = new Client({ auth: notionToken }); // 0. 公告欄 let announcements = []; if (announcementDbId) { try { const annRes = await notion.databases.query({ database_id: announcementDbId }); if (annRes?.results) { annRes.results.forEach(page => { const p = page?.properties || {}; const descKey = Object.keys(p).find(k => k && (k.includes('內容') || k.includes('說明') || k.includes('公告') || p[k]?.type === 'rich_text')); if (descKey && p[descKey]?.rich_text && Array.isArray(p[descKey].rich_text) && p[descKey].rich_text.length > 0) { const text = p[descKey].rich_text.map(t => t?.plain_text || '').join(''); if (text) announcements.push(text); } }); } } catch (err) { console.warn('公告讀取略過:', err.message); } } // 1. 查詢 A 表（主商品表） let validAMap = {}; if (productDbId) { try { const aRes = await notion.databases.query({ database_id: productDbId }); if (aRes?.results) { aRes.results.forEach(page => { const p = page?.properties || {}; let name = ''; const titleKey = Object.keys(p).find(k => k && p[k]?.type === 'title'); if (titleKey && p[titleKey]?.title && Array.isArray(p[titleKey].title) && p[titleKey].title.length > 0) { name = p[titleKey].title.map(t => t?.plain_text || '').join(''); } if (name) { let categories = ['伴手禮']; const catKey = Object.keys(p).find(k => k && (k.includes('分類') || k.includes('地區') || k.includes('Tag') || k.includes('分區'))); if (catKey && p[catKey]) { const prop = p[catKey]; if (prop.type === 'multi_select' && Array.isArray(prop.multi_select) && prop.multi_select.length > 0) { categories = prop.multi_select.map(item => item?.name).filter(Boolean); } else if (prop.type === 'select' && prop.select?.name) { categories = [prop.select.name]; } else if (prop.type === 'rich_text' && Array.isArray(prop.rich_text) && prop.rich_text.length > 0) { const txt = prop.rich_text.map(t => t?.plain_text || '').join('').trim(); if (txt) categories = [txt]; } } let needDeposit = false; const depositKey = Object.keys(p).find(k => k && (k.includes('訂金') || k.includes('押金'))); if (depositKey && p[depositKey]) { const prop = p[depositKey]; if (prop.type === 'checkbox') needDeposit = !!prop.checkbox; else if (prop.type === 'select') needDeposit = (prop.select?.name === '是' || prop.select?.name === '需訂金'); } validAMap[page.id] = { name, categories, needDeposit }; } }); } } catch (e) { console.warn('A表查詢略過:', e.message); } } // 2. 查詢 C 表（款式明細表） const cRes = await notion.databases.query({ database_id: productDetailDbId }); const products = []; if (cRes?.results) { cRes.results.forEach(page => { const p = page?.properties || {}; let saleType = '預購'; let isEnded = false; const statusKey = Object.keys(p).find(k => k && (k.includes('狀態') || p[k]?.type === 'status' || p[k]?.type === 'select' || p[k]?.type === 'formula')); if (statusKey && p[statusKey]) { const prop = p[statusKey]; let statusVal = ''; if (prop.type === 'status') statusVal = prop.status?.name || ''; else if (prop.type === 'select') statusVal = prop.select?.name || ''; else if (prop.type === 'formula') { if (prop.formula?.type === 'string') statusVal = prop.formula.string || ''; else if (prop.formula?.type === 'boolean') statusVal = prop.formula.boolean ? 'true' : 'false'; else if (prop.formula?.type === 'number') statusVal = String(prop.formula.number ?? ''); else statusVal = String(prop.formula?.string || prop.formula?.boolean || ''); } // 🔍 燈號與字串印出，讓妳在 Vercel Logs 裡看到真實抓到的值 console.log(`[Status Debug] 原始狀態值: "${statusVal}" (欄位: ${statusKey})`); // 只要狀態文字裡面包含這些關鍵字，直接判定為已結束！ const rawLower = String(statusVal).toLowerCase(); if ( rawLower.includes('結束') || rawLower.includes('結單') || rawLower.includes('斷貨') || rawLower.includes('售完') || rawLower.includes('已下架') || rawLower === 'false' || rawLower === 'closed' || rawLower === 'ended' ) { isEnded = true; saleType = '已結束'; } else if (statusVal.trim() !== '') { saleType = statusVal.replace(/^[🟢🔴🟡✅❌⛔✔️✖️\s]+/g, '').trim() || '預購'; } } if (isEnded) saleType = '已結束'; let parentName = '精選預購商品'; let categories = ['伴手禮']; let needDeposit = false; const relKey = Object.keys(p).find(k => k && p[k]?.type === 'relation'); if (relKey && p[relKey]?.relation && Array.isArray(p[relKey].relation) && p[relKey].relation.length > 0) { const relId = p[relKey].relation[0]?.id; if (relId && validAMap[relId]) { parentName = validAMap[relId].name; categories = validAMap[relId].categories; needDeposit = validAMap[relId].needDeposit; } } else { const titleKey = Object.keys(p).find(k => k && p[k]?.type === 'title'); if (titleKey && p[titleKey]?.title && Array.isArray(p[titleKey].title) && p[titleKey].title.length > 0) { parentName = p[titleKey].title.map(t => t?.plain_text || '').join('') || '精選預購商品'; } } let styleName = '標準款式'; const styleTitleKey = Object.keys(p).find(k => k && p[k]?.type === 'title'); if (styleTitleKey && p[styleTitleKey]?.title && Array.isArray(p[styleTitleKey].title) && p[styleTitleKey].title.length > 0) { styleName = styleTitleKey.title.map(t => t?.plain_text || '').join('') || '標準款式'; } let price = 0; const priceKey = Object.keys(p).find(k => k && (k.includes('價格') || k.includes('單價') || k.includes('Price')) && p[k]?.type === 'number'); if (priceKey && p[priceKey]?.number !== undefined) { price = p[priceKey].number || 0; } else { const anyNumKey = Object.keys(p).find(k => k && p[k]?.type === 'number'); if (anyNumKey && p[anyNumKey]?.number !== undefined) price = p[anyNumKey].number || 0; } let imageUrl = ''; const imgKey = Object.keys(p).find(k => k && (p[k]?.type === 'files' || k.includes('圖'))); if (imgKey && p[imgKey]?.files && Array.isArray(p[imgKey].files) && p[imgKey].files.length > 0) { const fileObj = imgKey.files[0]; if (fileObj) { imageUrl = fileObj.file?.url || fileObj.external?.url || ''; } } let deadline = ''; const dateKey = Object.keys(p).find(k => k && (k.includes('收單') || k.includes('日期') || k.includes('截止') || p[k]?.type === 'date')); if (dateKey && p[dateKey]?.date?.start) { deadline = p[dateKey].date.start; } products.push({ productName: parentName, categories: categories, saleType: saleType, deadline: deadline, needDeposit: needDeposit, styleName: styleName, price: price, imageUrl: imageUrl }); }); } return res.status(200).json({ success: true, announcements, products }); } catch (error) { console.error("Get Products Error:", error); return res.status(500).json({ success: false, error: error.message || String(error) }); } }
