function generateSidebarCategories(products) { const container = document.getElementById('sidebarCategories'); const allCats = []; products.forEach(p => { if (p.categories && Array.isArray(p.categories)) { allCats.push(...p.categories); } }); const uniqueCats = [...new Set(allCats)]; // 🌟 設定妳想要的自訂順序（只要字串包含這些關鍵字就會排在前面） const customOrder = ['新加坡', '伴手禮']; uniqueCats.sort((a, b) => { let indexA = customOrder.findIndex(keyword => a.includes(keyword)); let indexB = customOrder.findIndex(keyword => b.includes(keyword)); if (indexA === -1) indexA = 999; if (indexB === -1) indexB = 999; return indexA - indexB; }); const categories = ['全部', ...uniqueCats]; container.innerHTML = ''; categories.forEach(cat => { const btn = document.createElement('button'); btn.className = `sidebar-btn ${cat === currentCategory ? 'active' : ''}`; btn.innerText = cat; btn.onclick = () => filterCategory(cat, btn); container.appendChild(btn); }); }
